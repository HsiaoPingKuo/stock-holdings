  1. V4 完整 SQL 草稿

  檔名：src/main/resources/db/migration/V4__create_trade.sql

  -- 第 6 題：交易表。2.0 起持股改由「每次買賣」重播計算，1.0 
  的舊持股在這裡轉成一筆推估的買進
  -- 注意：這個檔案一旦 commit 就不能再改；之後資料表有任何變動，一律新增 
  V5__、V6__…

  CREATE TABLE trade (
      id              BIGINT GENERATED BY DEFAULT AS IDENTITY,
      -- 屬於哪一檔持股；holding 一個代號一筆，/api/v1 的 id 繼續用 holding.id
      holding_id      BIGINT         NOT NULL,
      -- 存 BUY 或 SELL（對應 Java enum 名稱）；欄位不叫 
  type，避免和日後除權息的「種類」混淆
      side            VARCHAR(4)     NOT NULL,
      -- 不叫 date：DATE 是 SQL 保留字
      trade_date      DATE           NOT NULL,
      -- 股數用 BIGINT：零股、超過 999 張、配股後都放得下（1.0 的 INT 張數上限是
  999）
      shares          BIGINT         NOT NULL,
      -- 和 holding.buy_price 一樣用 DECIMAL(12, 2)，對應 Java 的 BigDecimal
      price           DECIMAL(12, 2) NOT NULL,
      -- 日期是推估的（1.0 轉入、App 1.0 新增）時為 TRUE；使用者改過日期後改成 
  FALSE
      date_estimated  BOOLEAN        NOT NULL,
      -- 樂觀鎖的版本號：修改、刪除時比對，不一致代表別的裝置已經改過
      version         BIGINT         NOT NULL,
      -- 由程式用台北時區的 Clock 寫入，不依賴主機時區
      created_at      TIMESTAMP      NOT NULL,
      updated_at      TIMESTAMP      NOT NULL,
      -- 約束都自己取名字：日後要調整時，新的 migration 才能用名字 DROP 
  CONSTRAINT
      CONSTRAINT pk_trade         PRIMARY KEY (id),
      -- 刪除持股時一併刪除它的交易：現有的刪除功能（只刪 
  holding）才不會因為外鍵而失敗
      CONSTRAINT fk_trade_holding FOREIGN KEY (holding_id) REFERENCES holding
  (id) ON DELETE CASCADE,
      CONSTRAINT ck_trade_side    CHECK (side IN ('BUY', 'SELL')),
      CONSTRAINT ck_trade_shares  CHECK (shares > 0),
      CONSTRAINT ck_trade_price   CHECK (price > 0),
      -- 題目範圍從 2021 年開始；「不能是未來日期」要看今天，CHECK 
  做不到，由程式用 Clock 判斷
      CONSTRAINT ck_trade_date    CHECK (trade_date >= DATE '2021-01-01')
  );

  -- 最常用的查詢：某檔持股的全部交易，依日期、id 排序（重播順序）。
  -- holding_id 放第一欄，外鍵檢查和「刪持股時找出它的交易」也用得到這個索引
  CREATE INDEX idx_trade_holding_date ON trade (holding_id, trade_date, id);

  -- 1.0 的每筆持股轉成一筆買進：張數 × 1000 = 股數，買進價 = 成交價。
  -- 1.0 沒有記買進日期，用建立時間的日期推估（created_at 本來就是台北時間，CAST
  不會跨日），並標記為推估。
  -- 時間欄位沿用持股原本的值，不用 
  CURRENT_TIMESTAMP：那是主機時區，不是台北時間
  INSERT INTO trade (holding_id, side, trade_date, shares, price,
  date_estimated, version, created_at, updated_at)
  SELECT id, 'BUY', CAST(created_at AS DATE), CAST(lots AS BIGINT) * 1000,
  buy_price, TRUE, 0, created_at, updated_at
  FROM holding;

  -- 放寬 V1 的限制：2.0 的股數、成本改由交易計算，零股、0 張、超過 999 
  張都要能存。
  -- 舊欄位這一版先保留（擴充 → 遷移 → 收縮），之後再用新的 migration 刪除
  ALTER TABLE holding DROP CONSTRAINT ck_holding_lots;
  ALTER TABLE holding DROP CONSTRAINT ck_holding_buy_price;
  ALTER TABLE holding ALTER COLUMN lots DROP NOT NULL;
  ALTER TABLE holding ALTER COLUMN buy_price DROP NOT NULL;

  幾個設計說明：
  - version 和 date_estimated 都不給預設值。
    每次寫入都要明確決定這兩個值，就不會有「忘了設、被預設值偷偷帶過」的情況。
  - 轉入的交易，created_at、updated_at 沿用持股原本的值。 SQL 裡的
    CURRENT_TIMESTAMP 用的是主機時區（不是台北時間），會違反 CLAUDE.md
    的時間規則。
  - H2 執行 DDL（建表、改表這類指令）時不能整批回滾。 如果 V4
    跑到一半失敗，會留下半套的結構。所以下面的測試要確保 V4 在各種 1.0
    資料上都能跑完。

  2. 外鍵刪除方式：建議 ON DELETE CASCADE

  理由：
  1. 現有的刪除功能不能壞。 HoldingService.delete 只執行
     holdingRepository.delete(holding)（HoldingService.java:208）。V4
     之後，所有舊持股都有交易：
     - 如果用預設的 RESTRICT（有交易就不准刪持股），App 1.0
       和網頁刪除舊持股都會撞到外鍵錯誤，出現 500。
     - 這一階段又不能改 service。
  2. 交易離開持股就沒有意義。 不會有「持股刪了、交易要留著」的情況。由資料庫一次
     刪掉，一定是同時成功或同時失敗，也不怕之後某條刪除路徑忘了先刪交易。
  3. 「能不能刪」由 service 判斷，不靠外鍵擋。 決定 3 的 409
     UPGRADE_REQUIRED（有多筆交易時，App 1.0 不能刪）在第 3 階段由 service
     檢查。
     - 這一階段，App 1.0 刪除舊持股時會把那一筆推估買進一起刪掉。這和 1.0
       的行為一致（刪持股就是整筆不見）。

  代價與因應：
  - 資料庫自己刪掉交易時，JPA 不會知道。如果同一個交易裡先讀過 Trade
    再刪持股，記憶體裡會留著已刪除的物件。
  - 目前 Holding 沒有 @OneToMany 指向 Trade，這一階段也沒有程式讀
    Trade，所以不會發生。第 3、5 階段寫刪除時要留意這一點。

  沒採用「程式自己刪」的原因： 要改
  HoldingService.delete，超出這一階段的範圍；而且每條刪除路徑都要記得先刪交易。

  5. 第 2 階段合併後、第 3 階段完成前的不一致

  這段期間，holding 和 trade 會對不上：

  ┌────────────────────────┬───────────────────────────────────┐
  │ 操作（網頁或 App 1.0） │               結果                │
  ├────────────────────────┼───────────────────────────────────┤
  │ 新增持股               │ 只寫 holding，沒有對應的交易      │

  │ 刪除持股               │ 交易一起被刪（CASCADE），兩邊一致 │
  └────────────────────────┴───────────────────────────────────┘

  對使用者沒有影響： 這一階段沒有任何畫面或 API 讀 trade，看到的都還是 holding
  的 lots 和 buy_price。

  要注意的地方：
  1. 主考官不會受影響。 主考官是從 v1.0.0 的資料庫直接升級到交卷版本，V4
     跑的時候會把當下所有持股完整轉入，中間沒有空窗。
  2. 你自己的 ./data/ 會受影響。 這段期間如果用網頁新增或修改持股，V4
     已經跑過、不會再跑。第 3 階段改成讀 trade 之後，這些持股會變成：
     - 沒有交易 → 0 股，從清單消失
     - 或是數字是舊的
  3. 建議： 第 3 階段開始前，先刪掉或備份
     ./data/。之後「升級前後回應一樣」的手動驗證，改用 v1.0.0 
     建出來的乾淨資料庫，不要用開發中的 ./data/。
  4. 不建議為了這段空窗寫一個 V5 補交易。這只影響開發者自己的資料庫，加一個
     migration 反而讓升級路徑變複雜。

  6. 需要你決定的一件事

  要不要對「早於 2021-01-01 的建立日期」加保護？
  - ck_trade_date 會擋掉早於 2021-01-01 的日期。如果某個 1.0 資料庫裡的
    created_at 早於 2021，V4 會失敗，伺服器就啟動不了。
  - 正常使用不會發生（1.0 是 2026 年寫的，時間由 Clock
    寫入）；只有人手動改過資料庫才會。
  - 選項：
    - A（照你給的規格）： CAST(created_at AS DATE)，不額外處理。
    - B： 改成 GREATEST(CAST(created_at AS DATE), DATE 
      '2021-01-01')，最壞情況日期被拉到 2021-01-01，但一定啟動得了。
  - 我傾向 A。 這種資料實際上不存在，多一行反而要多解釋一次；不過「啟動不了就零
    分」，所以交給你決定。

  -- 第 6 題：交易表。2.0 起持股改由「每次買賣」重播計算，1.0 
  的舊持股在這裡轉成一筆推估的買進
  -- 注意：這個檔案一旦 commit 就不能再改；之後資料表有任何變動，一律新增 
  V5__、V6__…

  CREATE TABLE trade (
      id              BIGINT GENERATED BY DEFAULT AS IDENTITY,
      -- 屬於哪一檔持股；holding 一個代號一筆，/api/v1 的 id 繼續用 holding.id
      holding_id      BIGINT         NOT NULL,
      -- 存 BUY 或 SELL（對應 Java enum 名稱）；欄位不叫 
  type，避免和日後除權息的「種類」混淆
      side            VARCHAR(4)     NOT NULL,
      -- 不叫 date：DATE 是 SQL 保留字
      trade_date      DATE           NOT NULL,
      -- 股數用 BIGINT：零股、超過 999 張、配股後都放得下（1.0 的 INT 張數上限是
  999）
      shares          BIGINT         NOT NULL,
      -- 和 holding.buy_price 一樣用 DECIMAL(12, 2)，對應 Java 的 BigDecimal
      price           DECIMAL(12, 2) NOT NULL,
      -- 日期是推估的（1.0 轉入、App 1.0 新增）時為 TRUE；使用者改過日期後改成 
  FALSE
      date_estimated  BOOLEAN        NOT NULL,
      -- 樂觀鎖的版本號：修改、刪除時比對，不一致代表別的裝置已經改過
      version         BIGINT         NOT NULL,
      -- 由程式用台北時區的 Clock 寫入，不依賴主機時區
      created_at      TIMESTAMP      NOT NULL,
      updated_at      TIMESTAMP      NOT NULL,
      -- 約束都自己取名字：日後要調整時，新的 migration 才能用名字 DROP 
  CONSTRAINT
      CONSTRAINT pk_trade         PRIMARY KEY (id),
      -- 刪除持股時一併刪除它的交易：現有的刪除功能（只刪 
  holding）才不會因為外鍵而失敗
      CONSTRAINT fk_trade_holding FOREIGN KEY (holding_id) REFERENCES holding
  (id) ON DELETE CASCADE,
      CONSTRAINT ck_trade_side    CHECK (side IN ('BUY', 'SELL')),
      CONSTRAINT ck_trade_shares  CHECK (shares > 0),
      CONSTRAINT ck_trade_price   CHECK (price > 0),
      -- 題目範圍從 2021 年開始；「不能是未來日期」要看今天，CHECK 
  做不到，由程式用 Clock 判斷
      CONSTRAINT ck_trade_date    CHECK (trade_date >= DATE '2021-01-01')
  );

  -- 最常用的查詢：某檔持股的全部交易，依日期、id 排序（重播順序）。
  -- holding_id 放第一欄，外鍵檢查和「刪持股時找出它的交易」也用得到這個索引
  CREATE INDEX idx_trade_holding_date ON trade (holding_id, trade_date, id);

  -- 1.0 的每筆持股轉成一筆買進：張數 × 1000 = 股數，買進價 = 成交價。
  -- 1.0 沒有記買進日期，用建立時間的日期推估（created_at 本來就是台北時間，CAST
  不會跨日），並標記為推估。
  -- GREATEST：建立日期早於 2021-01-01 
  時（正常使用不會發生，只有手動改過資料庫才會）拉到 2021-01-01。
  -- 轉入的日期本來就是推估，拉到最早日期一樣標示推估；但保證不管什麼資料庫，V4 
  都不會被 ck_trade_date 擋下而無法啟動
  -- 時間欄位沿用持股原本的值，不用 
  CURRENT_TIMESTAMP：那是主機時區，不是台北時間
  INSERT INTO trade (holding_id, side, trade_date, shares, price,
  date_estimated, version, created_at, updated_at)
  SELECT id, 'BUY', GREATEST(CAST(created_at AS DATE), DATE '2021-01-01'),
  CAST(lots AS BIGINT) * 1000, buy_price,
         TRUE, 0, created_at, updated_at
  FROM holding;

  -- 放寬 V1 的限制：2.0 的股數、成本改由交易計算，零股、0 張、超過 999 
  張都要能存。
  -- 舊欄位這一版先保留（擴充 → 遷移 → 收縮），之後再用新的 migration 刪除
  ALTER TABLE holding DROP CONSTRAINT ck_holding_lots;
  ALTER TABLE holding DROP CONSTRAINT ck_holding_buy_price;
  ALTER TABLE holding ALTER COLUMN lots DROP NOT NULL;
  ALTER TABLE holding ALTER COLUMN buy_price DROP NOT NULL;
